-- 1. Borrar todo lo anterior (si existe), en orden por dependencias
DROP TABLE IF EXISTS oficial.examen CASCADE;
DROP TABLE IF EXISTS oficial.equivalencia CASCADE;
DROP TABLE IF EXISTS oficial.nota CASCADE;
DROP TABLE IF EXISTS oficial.cursada CASCADE;
DROP TABLE IF EXISTS oficial.permiso CASCADE;
DROP SCHEMA IF EXISTS oficial CASCADE;

-- 2. Crear el esquema
CREATE SCHEMA oficial;

-- 3. Crear las tablas copiando estructura de public (columnas, defaults, constraints, índices)
CREATE TABLE oficial.permiso      (LIKE public.permiso      INCLUDING ALL);
CREATE TABLE oficial.cursada      (LIKE public.cursada      INCLUDING ALL);
CREATE TABLE oficial.nota         (LIKE public.nota         INCLUDING ALL);
CREATE TABLE oficial.examen       (LIKE public.examen       INCLUDING ALL);
CREATE TABLE oficial.equivalencia (LIKE public.equivalencia INCLUDING ALL);

-- 4. FKs internas al esquema oficial
ALTER TABLE oficial.nota
  ADD CONSTRAINT fk_nota_cursada FOREIGN KEY (nota_cursada_id) REFERENCES oficial.cursada(cursada_id);

ALTER TABLE oficial.examen
  ADD CONSTRAINT fk_examen_nota            FOREIGN KEY (nota_id)            REFERENCES oficial.nota(nota_id),
  ADD CONSTRAINT fk_examen_permiso         FOREIGN KEY (permiso_id)         REFERENCES oficial.permiso(permiso_id),
  ADD CONSTRAINT fk_examen_cursada_examen  FOREIGN KEY (cursada_examen_id)  REFERENCES public.cursada_examen(cursada_examen_id);

ALTER TABLE oficial.equivalencia
  ADD CONSTRAINT fk_equivalencia_nota FOREIGN KEY (nota_id) REFERENCES oficial.nota(nota_id);

-- 5. FKs de oficial.cursada hacia las tablas maestras que siguen en public
ALTER TABLE oficial.cursada
  ADD CONSTRAINT fk_cursada_materia_carrera FOREIGN KEY (cursada_materia_carrera_id) REFERENCES public.materia_carrera(id);
  -- + FK a legajo/alumno si corresponde, ej:
